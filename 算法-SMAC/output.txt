def save_paras_metric(paras: dict, metric, file_path):
    # 读取现有文件内容
    try:
        with open(file_path, "r", encoding="utf-8") as file:
            lines = file.readlines()
    except FileNotFoundError:
        lines = []

    # 准备新数据
    new_data = [f"{key}\t{value}" for key, value in paras.items()] + [f"metric\t{metric}"]

    # 处理文件为空的情况
    if not lines:
        lines = ['' for _ in range(len(new_data))]

    # 确保行数一致
    while len(lines) < len(new_data):
        lines.append("\n")

    # 追加新数据到每一行的末尾
    updated_lines = []
    for line, new_line in zip(lines, new_data):
        updated_lines.append(line.strip() + "\t" + new_line + "\n")

    # 将更新后的内容写回文件
    with open(file_path, "w", encoding="utf-8") as file:
        file.writelines(updated_lines)
